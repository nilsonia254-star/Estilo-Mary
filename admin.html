function sendWhatsApp() { const c = clientsList[currentIndex]; if(!c.phone) { alert("Esta cliente não tem número de WhatsApp cadastrado!"); return; } // Limpa o número para conter apenas dígitos (remove espaços, traços, parênteses) let telefoneLimpo = c.phone.replace(/\D/g, ''); // Se o número não começar com o código do país (55), adiciona o 55 do Brasil if (!telefoneLimpo.startsWith('55')) { telefoneLimpo = '55' + telefoneLimpo; } const dates = getClientCardDates(c); // Mensagem formatada para a cliente const textoMensagem = `Olá *${c.name}*! ✨ Aqui está o status atualizado do seu Cartão Fidelidade no *Designer Estilo Mary*:\n\n⭐ ID: ${c.id}\n🎁 Seus Pontos: ${c.points} pts\n📅 Válido até: ${dates.expiryFormatted} (${dates.daysLeft} dias restantes)\n\nVenha nos visitar e acumule mais pontos!`; const msgCodificada = encodeURIComponent(textoMensagem); // Abre o WhatsApp Web ou o App do celular diretamente com a mensagem pronta const urlWhatsApp = `https://wa.me/${telefoneLimpo}?text=${msgCodificada}`; window.open(urlWhatsApp, '_blank'); }
