function addPoints(pts) { if(clientsList.length > 0) { clientsList[currentIndex].name = document.getElementById('clientName').value || clientsList[currentIndex].name; clientsList[currentIndex].points = Math.max(0, clientsList[currentIndex].points + pts); // ATENÇÃO: Removemos a sobrescrita do lastUpdate aqui para a data do atendimento original não mudar sozinha todos os dias! renderTableAndSelect(); selectClientByIndex(currentIndex); } } function resetPoints() { if(clientsList.length > 0) { clientsList[currentIndex].points = 0; // Mantém a data original do cadastro/atendimento intacta renderTableAndSelect(); selectClientByIndex(currentIndex); } } function adjustDays(days) { if(clientsList.length > 0) { clientsList[currentIndex].days = Math.max(0, (clientsList[currentIndex].days || 60) + days); document.getElementById('validityDays').value = clientsList[currentIndex].days; renderTableAndSelect(); selectClientByIndex(currentIndex); } } function saveValidityDays() { if(clientsList.length > 0) { const val = parseInt(document.getElementById('validityDays').value) || 60; clientsList[currentIndex].days = val; // Mantém a data original fixa, alterando apenas os dias de validade renderTableAndSelect(); alert("Validade de " + val + " dias fixada!"); } }
